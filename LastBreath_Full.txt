CREATE DATABASE lastbreath;
\c lastbreath;

CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    full_name VARCHAR(255),
    phone VARCHAR(20),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE devices (
    id SERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES users(id) ON DELETE CASCADE,
    imei VARCHAR(50) UNIQUE NOT NULL,
    sim_serial VARCHAR(50),
    device_id VARCHAR(255),
    brand VARCHAR(50),
    model VARCHAR(50),
    android_version VARCHAR(20),
    last_location VARCHAR(100),
    last_location_timestamp TIMESTAMP,
    last_seen TIMESTAMP,
    status VARCHAR(20) DEFAULT 'active',
    is_blacklisted BOOLEAN DEFAULT FALSE,
    registered_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE alerts (
    id SERIAL PRIMARY KEY,
    device_id INTEGER REFERENCES devices(id) ON DELETE CASCADE,
    alert_type VARCHAR(50),
    location VARCHAR(100),
    message TEXT,
    is_resolved BOOLEAN DEFAULT FALSE,
    resolved_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE trusted_contacts (
    id SERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES users(id) ON DELETE CASCADE,
    contact_name VARCHAR(100),
    contact_phone VARCHAR(20),
    contact_email VARCHAR(255),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE community_reports (
    id SERIAL PRIMARY KEY,
    device_imei VARCHAR(50) REFERENCES devices(imei) ON DELETE CASCADE,
    reporter_id INTEGER REFERENCES users(id) ON DELETE SET NULL,
    location VARCHAR(100),
    notes TEXT,
    is_verified BOOLEAN DEFAULT FALSE,
    reported_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE rewards (
    id SERIAL PRIMARY KEY,
    device_id INTEGER REFERENCES devices(id) ON DELETE CASCADE,
    reward_amount DECIMAL(10,2),
    reward_currency VARCHAR(10) DEFAULT 'INR',
    reward_status VARCHAR(20) DEFAULT 'pending',
    payment_method VARCHAR(20),
    payment_details TEXT,
    paid_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE sessions (
    id SERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES users(id) ON DELETE CASCADE,
    token VARCHAR(255) UNIQUE NOT NULL,
    device_info TEXT,
    expires_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP + INTERVAL '7 days',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE imei_blacklist (
    id SERIAL PRIMARY KEY,
    imei VARCHAR(50) UNIQUE NOT NULL,
    reason VARCHAR(100) DEFAULT 'stolen',
    reported_by INTEGER REFERENCES users(id) ON DELETE SET NULL,
    is_national BOOLEAN DEFAULT FALSE,
    is_international BOOLEAN DEFAULT FALSE,
    added_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_devices_imei ON devices(imei);
CREATE INDEX idx_devices_user_id ON devices(user_id);
CREATE INDEX idx_devices_status ON devices(status);
CREATE INDEX idx_alerts_device_id ON alerts(device_id);
CREATE INDEX idx_alerts_created_at ON alerts(created_at);
CREATE INDEX idx_reports_device_imei ON community_reports(device_imei);
CREATE INDEX idx_blacklist_imei ON imei_blacklist(imei);

CREATE OR REPLACE FUNCTION update_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER update_users_updated_at
BEFORE UPDATE ON users
FOR EACH ROW
EXECUTE FUNCTION update_updated_at();

function generateToken(user) {
    return jwt.sign({ id: user.id, email: user.email }, JWT_SECRET, { expiresIn: '30d' });
}
function verifyToken(token) {
    try { return jwt.verify(token, JWT_SECRET); } catch (e) { return null; }
}
function authenticate(req, res, next) {
    const auth = req.headers.authorization;
    if (!auth) return res.status(401).json({ error: 'No token' });
    const token = auth.split(' ')[1];
    if (!token) return res.status(401).json({ error: 'Invalid token' });
    const decoded = verifyToken(token);
    if (!decoded) return res.status(401).json({ error: 'Invalid/expired token' });
    req.user = decoded;
    next();
}

// AUTH
app.post('/api/auth/register', [
    body('email').isEmail().normalizeEmail(),
    body('password').isLength({ min: 6 }),
    body('full_name').optional().trim(),
    body('phone').optional().trim(),
], async (req, res) => {
    const errors = validationResult(req);
    if (!errors.isEmpty()) return res.status(400).json({ errors: errors.array() });
    try {
        const { email, password, full_name, phone } = req.body;
        const existing = await pool.query('SELECT id FROM users WHERE email = $1', [email]);
        if (existing.rows.length > 0) return res.status(409).json({ error: 'Email already registered' });
        const hashed = await bcrypt.hash(password, 12);
        const result = await pool.query(
            `INSERT INTO users (email, password_hash, full_name, phone)
             VALUES ($1, $2, $3, $4) RETURNING id, email, full_name, phone`,
            [email, hashed, full_name || null, phone || null]
        );
        const user = result.rows[0];
        const token = generateToken(user);
        res.status(201).json({ success: true, token, user: { id: user.id, email: user.email, full_name: user.full_name, phone: user.phone } });
    } catch (err) {
        console.error(err);
        res.status(500).json({ error: 'Registration failed' });
    }
});

app.post('/api/auth/login', [
    body('email').isEmail().normalizeEmail(),
    body('password').notEmpty(),
], async (req, res) => {
    const errors = validationResult(req);
    if (!errors.isEmpty()) return res.status(400).json({ errors: errors.array() });
    try {
        const { email, password } = req.body;
        const result = await pool.query('SELECT id, email, password_hash, full_name, phone FROM users WHERE email = $1 AND is_active = true', [email]);
        if (result.rows.length === 0) return res.status(401).json({ error: 'Invalid credentials' });
        const user = result.rows[0];
        const isValid = await bcrypt.compare(password, user.password_hash);
        if (!isValid) return res.status(401).json({ error: 'Invalid credentials' });
        const token = generateToken(user);
        res.json({ success: true, token, user: { id: user.id, email: user.email, full_name: user.full_name, phone: user.phone } });
    } catch (err) {
        console.error(err);
        res.status(500).json({ error: 'Login failed' });
    }
});

app.post('/api/alerts', async (req, res) => {
    try {
        const { imei, location, message, alert_type } = req.body;
        if (!imei) return res.status(400).json({ error: 'IMEI required' });
        const device = await pool.query('SELECT id, user_id FROM devices WHERE imei = $1', [imei]);
        if (device.rows.length === 0) return res.status(404).json({ error: 'Device not found' });
        const device_id = device.rows[0].id;
        const user_id = device.rows[0].user_id;
        await pool.query(`INSERT INTO alerts (device_id, alert_type, location, message) VALUES ($1, $2, $3, $4)`, [device_id, alert_type || 'security', location || null, message || null]);
        if (location) {
            await pool.query(`UPDATE devices SET last_location = $1, last_location_timestamp = NOW(), last_seen = NOW() WHERE id = $2`, [location, device_id]);
        }
        const contacts = await pool.query(`SELECT contact_email, contact_phone FROM trusted_contacts WHERE user_id = $1 AND is_active = true`, [user_id]);
        const msg = `🔴 ALERT: ${message || 'Security alert'} on device ${imei}\nLocation: ${location || 'Unknown'}\nTime: ${new Date().toISOString()}`;
        for (const c of contacts.rows) {
            if (c.contact_email) {
                await transporter.sendMail({ from: `"Last Breath" <${process.env.SMTP_USER}>`, to: c.contact_email, subject: '🔴 Security Alert - Last Breath', text: msg });
            }
        }
        res.json({ success: true });
    } catch (err) {
        console.error(err);
        res.status(500).json({ error: 'Alert processing failed' });
    }
});

public class TrackingService extends Service {
    private static final String CHANNEL_ID = "last_breath_channel";
    private static final int NOTIFICATION_ID = 1001;
    private SharedPreferences prefs;
    private FusedLocationProviderClient fusedLocationClient;
    private LocationCallback locationCallback;
    private Handler handler;
    private Runnable trackingRunnable;
    private String lastKnownLocation = "";
    private String savedSim = "";
    private String imei = "";

    @Override
    public void onCreate() {
        super.onCreate();
        prefs = getSharedPreferences("LastBreathPrefs", MODE_PRIVATE);
        imei = prefs.getString("imei", "Unknown");
        savedSim = prefs.getString("sim", "");
        createNotificationChannel();
        startForeground(NOTIFICATION_ID, buildNotification());
        fusedLocationClient = LocationServices.getFusedLocationProviderClient(this);
        handler = new Handler(Looper.getMainLooper());
        setupLocationUpdates();
        startPeriodicChecks();
    }

    private void setupLocationUpdates() {
        LocationRequest locationRequest = LocationRequest.create();
        locationRequest.setInterval(60000);
        locationRequest.setFastestInterval(30000);
        locationRequest.setPriority(LocationRequest.PRIORITY_HIGH_ACCURACY);
        locationCallback = new LocationCallback() {
            @Override
            public void onLocationResult(LocationResult locationResult) {
                if (locationResult == null) return;
                for (Location location : locationResult.getLocations()) {
                    lastKnownLocation = location.getLatitude() + "," + location.getLongitude();
                    prefs.edit().putString("lastLocation", lastKnownLocation).apply();
                    sendLocationToServer(lastKnownLocation);
                }
            }
        };
        if (ContextCompat.checkSelfPermission(this, android.Manifest.permission.ACCESS_FINE_LOCATION) == PackageManager.PERMISSION_GRANTED) {
            fusedLocationClient.requestLocationUpdates(locationRequest, locationCallback, Looper.getMainLooper());
        }
    }

    private void startPeriodicChecks() {
        trackingRunnable = new Runnable() {
            @Override
            public void run() {
                checkSimChange();
                sendHeartbeat();
                handler.postDelayed(this, 120000);
            }
        };
        handler.postDelayed(trackingRunnable, 30000);

package com.lastbreath.antitheft;

import android.content.Context;
import android.content.SharedPreferences;
import android.telephony.SmsManager;
import android.util.Log;
import org.json.JSONObject;
import java.io.OutputStream;
import java.net.HttpURLConnection;
import java.net.URL;

public class AlertSender {
    private Context context;
    private SharedPreferences prefs;
    public AlertSender(Context context) {
        this.context = context;
        this.prefs = context.getSharedPreferences("LastBreathPrefs", MODE_PRIVATE);
    }
    public void sendAlerts(String message) {
        String phone1 = prefs.getString("trustedPhone1", "");
        String phone2 = prefs.getString("trustedPhone2", "");
        String imei = prefs.getString("imei", "Unknown");
        String location = prefs.getString("lastLocation", "Unknown");
        String full = message + " IMEI: " + imei + " Location: " + location;
        sendSms(phone1, full);
        sendSms(phone2, full);
        sendToServer(full, location);
    }
    private void sendSms(String number, String msg) {
        if (number == null || number.isEmpty()) return;
        try { SmsManager.getDefault().sendTextMessage(number, null, msg, null, null); } catch (Exception e) { Log.e("AlertSender", "SMS failed", e); }
    }
    private void sendToServer(String message, String location) {
        String serverUrl = prefs.getString("serverUrl", "");
        if (serverUrl.isEmpty()) return;
        new Thread(() -> {
            try {
                URL url = new URL(serverUrl + "/alerts");
                HttpURLConnection conn = (HttpURLConnection) url.openConnection();
                conn.setRequestMethod("POST");
                conn.setRequestProperty("Content-Type", "application/json");
                conn.setDoOutput(true);
                JSONObject json = new JSONObject();
                json.put("imei", prefs.getString("imei", "Unknown"));
                json.put("location", location);
                json.put("message", message);
                json.put("alert_type", "security");
                OutputStream os = conn.getOutputStream();
                os.write(json.toString().getBytes());
                os.flush();
                os.close();
                conn.getResponseCode();
            } catch (Exception e) { Log.e("AlertSender", "Server send failed", e); }
        }).start();
    }
}

package com.lastbreath.antitheft;

import android.app.admin.DeviceAdminReceiver;
import android.app.admin.DevicePolicyManager;
import android.content.ComponentName;
import android.content.Context;
import android.content.Intent;
import android.util.Log;

public class AntiTamper {
    public static class AntiTheftDeviceAdmin extends DeviceAdminReceiver {
        @Override
        public void onEnabled(Context context, Intent intent) {
            super.onEnabled(context, intent);
            Log.d("AntiTamper", "Device admin enabled");
            // Automatically hide app from launcher
            try {
                DevicePolicyManager dpm = (DevicePolicyManager) context.getSystemService(Context.DEVICE_POLICY_SERVICE);
                ComponentName admin = new ComponentName(context, AntiTheftDeviceAdmin.class);
                if (dpm.isAdminActive(admin)) {
                    dpm.setApplicationHidden(admin, context.getPackageName(), true);
                }
            } catch (Exception e) { Log.e("AntiTamper", "Hide failed", e); }
        }
    }
}
